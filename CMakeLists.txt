cmake_minimum_required(VERSION 3.21)
project(PCLQt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Network Concurrent)
qt_standard_project_setup(REQUIRES 6.8)
if(MINGW)
    get_filename_component(MINGW_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(MINGW_ROOT "${MINGW_BIN}" DIRECTORY)
    if(EXISTS "${MINGW_ROOT}/x86_64-w64-mingw32/include/zlib.h")
        set(ZLIB_INCLUDE_DIR "${MINGW_ROOT}/x86_64-w64-mingw32/include")
        set(ZLIB_LIBRARY "${MINGW_ROOT}/x86_64-w64-mingw32/lib/libz.a")
    endif()
endif()
find_package(ZLIB REQUIRED)
add_executable(PCLQt WIN32 main.cpp launcher_controller.cpp launcher_controller.h instance_store.cpp instance_store.h launcher.cpp launcher.h version_model.cpp version_model.h version_status.cpp version_status.h java_discovery.cpp java_discovery.h)
set_source_files_properties(qml/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
qt_add_qml_module(PCLQt URI PCLQt VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(PCLQt PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent ZLIB::ZLIB)
